DROP DATABASE IF EXISTS petshop_db;

CREATE DATABASE petshop_db;

USE petshop_db;

CREATE TABLE funcionario (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50),
    apellido VARCHAR(50),
    cedula VARCHAR(20),
    telefono VARCHAR(20),
    legajo VARCHAR(20),
    rol VARCHAR(20),
    usuario VARCHAR(30),
    contrasena VARCHAR(50)
);

CREATE TABLE cliente (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50),
    apellido VARCHAR(50),
    cedula VARCHAR(20),
    telefono VARCHAR(20)
);

CREATE TABLE mascota (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50),
    especie VARCHAR(30),
    raza VARCHAR(30),
    id_cliente INT,
    FOREIGN KEY (id_cliente) REFERENCES cliente(id)
);

CREATE TABLE turno (
    id INT PRIMARY KEY AUTO_INCREMENT,
    id_mascota INT,
    tipo_servicio VARCHAR(20),
    fecha DATE,
    hora TIME,
    estado VARCHAR(20),
    id_banador INT,
    FOREIGN KEY (id_mascota) REFERENCES mascota(id),
    FOREIGN KEY (id_banador) REFERENCES funcionario(id)
);

CREATE TABLE consulta (
    id INT PRIMARY KEY AUTO_INCREMENT,
    id_cliente INT,
    id_funcionario INT,
    descripcion VARCHAR(255),
    fecha DATE,
    FOREIGN KEY (id_cliente) REFERENCES cliente(id),
    FOREIGN KEY (id_funcionario) REFERENCES funcionario(id)
);
